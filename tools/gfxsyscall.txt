
// region GDI
//
// =====================================================
//  Graphics Subsystem (GFX)
//  Syscalls 300–399
// =====================================================
//

// -----------------------------------------------------
//  Type Aliases
// -----------------------------------------------------

using HANDLER = c<uint32_t>;
using COLORREF = c<uint32_t>;
// COLORREF is a palette16 index (0..15), shared with USER class backgrounds.
// Packed RGB targets receive the corresponding canonical VGA RGB color.
using POINT = struct<Point>;
using RECT  = struct<Rect>;


// -----------------------------------------------------
//  Flag Types
// -----------------------------------------------------

flags DCInitFlags : uint32 {
    Primary;
    Memory;
    Offscreen;
    NoAccel;
};

flags PenStyle : uint32 {
    Solid;
    Dash;
    Dot;
    DashDot;
    DashDotDot;
};

flags BrushPattern : uint32 {
    Solid;
    Hatch;
    Diagonal;
    Cross;
};

flags RegionCombineMode : uint32 {
    Union;
    Intersect;
    Subtract;
    Xor;
};

flags Device3DFlags : uint32 {
    Hardware;
    Software;
    VertexAccel;
    TextureAccel;
};


// -----------------------------------------------------
//  Struct Types
// -----------------------------------------------------

struct Point {
    x: c<int32_t>;
    y: c<int32_t>;
};

struct Rect {
    left:   c<int32_t>;
    top:    c<int32_t>;
    right:  c<int32_t>;
    bottom: c<int32_t>;
};

struct DeviceCaps {
    width:      c<uint32_t>;
    height:     c<uint32_t>;
    colorDepth: c<uint32_t>;
    accelFlags: c<uint32_t>;
};

struct TextMetrics {
    cellWidth:   c<uint32_t>;
    cellHeight:  c<uint32_t>;
    ascent:      c<uint32_t>;
    descent:     c<uint32_t>;
    firstChar:   c<uint32_t>;
    lastChar:    c<uint32_t>;
    defaultChar: c<uint32_t>;
};


// -----------------------------------------------------
//  Pixel Formats & Raster Ops (Enums)
// -----------------------------------------------------

enum PixelFormat {
    RGB24  = 0,
    ARGB32 = 1,
    INDEX8 = 2,
};

enum RasterOp {
    SRCCOPY    = 0,
    SRCAND     = 1,
    SRCINVERT  = 2,
    SRCERASE   = 3,
    NOTSRCCOPY = 4,
};


// -----------------------------------------------------
//  Device Context & Object Lifecycle
// -----------------------------------------------------

// Zero/Primary selects the screen. Memory/Offscreen creates a DC with no target
// until SelectObject selects a bitmap. Primary cannot combine with these flags.
// NoAccel bypasses packed acceleration, never the required planar adapter path.
syscall V1 GFX CreateDC [300] {
    driverName: IN string;
    deviceName: IN string;
    initFlags:  IN flags<DCInitFlags>;
    handle:     OUT c<uint32_t>;
};

syscall V1 GFX DeleteDC [301] {
    dc: IN HANDLER;
};

// Only bitmap selection into Memory/Offscreen DCs is supported.
syscall V1 GFX SelectObject [302] {
    dc:     IN HANDLER;
    object: IN HANDLER;
};

syscall V1 GFX CreatePen [303] {
    style: IN flags<PenStyle>;
    width: IN c<uint32_t>;
    color: IN COLORREF;
};

syscall V1 GFX CreateBrush [304] {
    pattern: IN flags<BrushPattern>;
    color:   IN COLORREF;
};

syscall V1 GFX CreateFont [305] {
    height:  IN c<int32_t>;
    weight:  IN c<uint32_t>;
    charset: IN c<uint32_t>;
    name:    IN string;
};

// Top-down packed bitmap, zero initialized; maximum backing allocation 16 MiB.
// INDEX8 uses palette16 indices; RGB24/ARGB32 stores native BGR/BGRA bytes.
syscall V1 GFX CreateBitmap [306] {
    width:   IN c<uint32_t>;
    height:  IN c<uint32_t>;
    format:  IN enum<PixelFormat>;
    pixels:  IN buffer<c<uint8_t>>;
    handle:  OUT c<uint32_t>;
};

syscall V1 GFX DeleteObject [307] {
    object: IN HANDLER;
};


// -----------------------------------------------------
//  Drawing Primitives
// -----------------------------------------------------

syscall V1 GFX MoveTo [310] {
    dc: IN HANDLER;
    x:  IN c<int32_t>;
    y:  IN c<int32_t>;
};

syscall V1 GFX LineTo [311] {
    dc: IN HANDLER;
    x:  IN c<int32_t>;
    y:  IN c<int32_t>;
};

syscall V1 GFX Rectangle [312] {
    dc:     IN HANDLER;
    left:   IN c<int32_t>;
    top:    IN c<int32_t>;
    right:  IN c<int32_t>;
    bottom: IN c<int32_t>;
};

syscall V1 GFX Ellipse [313] {
    dc:     IN HANDLER;
    left:   IN c<int32_t>;
    top:    IN c<int32_t>;
    right:  IN c<int32_t>;
    bottom: IN c<int32_t>;
};

// Original fixed 8x16 single-line ASCII32..126; other bytes render '?'.
// No wrapping, shaping, font objects, newlines or cursor advancement.
syscall V1 GFX TextOut [314] {
    dc:   IN HANDLER;
    x:    IN c<int32_t>;
    y:    IN c<int32_t>;
    text: IN string;
};


// -----------------------------------------------------
//  Blits & Raster Ops
// -----------------------------------------------------

// Only SRCCOPY is supported; source/destination are clipped in lockstep.
// Planar-to-packed transfers are unsupported (no unsafe packed fallback).
syscall V1 GFX BitBlt [320] {
    dstDC:   IN HANDLER;
    x:       IN c<int32_t>;
    y:       IN c<int32_t>;
    width:   IN c<uint32_t>;
    height:  IN c<uint32_t>;
    srcDC:   IN HANDLER;
    srcX:    IN c<int32_t>;
    srcY:    IN c<int32_t>;
    rop:     IN enum<RasterOp>;
};

syscall V1 GFX StretchBlt [321] {
    dstDC:     IN HANDLER;
    x:         IN c<int32_t>;
    y:         IN c<int32_t>;
    dstWidth:  IN c<uint32_t>;
    dstHeight: IN c<uint32_t>;
    srcDC:     IN HANDLER;
    srcX:      IN c<int32_t>;
    srcY:      IN c<int32_t>;
    srcWidth:  IN c<uint32_t>;
    srcHeight: IN c<uint32_t>;
    rop:       IN enum<RasterOp>;
};

syscall V1 GFX AlphaBlend [322] {
    dstDC:   IN HANDLER;
    srcDC:   IN HANDLER;
    dstRect: IN RECT;
    srcRect: IN RECT;
    alpha:   IN c<uint8_t>;
};


// -----------------------------------------------------
//  Regions & State
// -----------------------------------------------------

syscall V1 GFX CreateRegion [330] {
    rects: IN dynarray<RECT>;
};

syscall V1 GFX SelectClipRegion [331] {
    dc:     IN HANDLER;
    region: IN HANDLER;
};

syscall V1 GFX CombineRegion [332] {
    dest: IN HANDLER;
    src1: IN HANDLER;
    src2: IN HANDLER;
    mode: IN flags<RegionCombineMode>;
};

syscall V1 GFX SetViewportOrg [333] {
    dc: IN HANDLER;
    x:  IN c<int32_t>;
    y:  IN c<int32_t>;
};

syscall V1 GFX SetWindowOrg [334] {
    dc: IN HANDLER;
    x:  IN c<int32_t>;
    y:  IN c<int32_t>;
};


// -----------------------------------------------------
//  Metrics & Queries
// -----------------------------------------------------

syscall V1 GFX GetTextMetrics [340] {
    dc:      IN HANDLER;
    metrics: OUT struct<TextMetrics>;
};

syscall V1 GFX GetBitmapBits [341] {
    bitmap: IN HANDLER;
    buffer: OUT buffer<c<uint8_t>>;
};

syscall V1 GFX SetBitmapBits [342] {
    bitmap: IN HANDLER;
    buffer: IN buffer<c<uint8_t>>;
};

syscall V1 GFX GetDeviceCaps [343] {
    dc:   IN HANDLER;
    caps: OUT struct<DeviceCaps>;
};

// Sets Rectangle/LineTo/TextOut foreground and TextOut background.
// opaque is 0 (transparent) or 1 (fill each complete 8x16 character cell).
syscall V1 GFX SetDrawColors [344] {
    dc:         IN c<uint32_t>;
    foreground: IN COLORREF;
    background: IN COLORREF;
    opaque:     IN c<uint32_t>;
};

// Normalized UFT1 resources only, never original TTF bytes. Font resource
// handles are process-local and distinct from DC/general object handles.
// Capability bit 0: private resource registration; no rendering bit yet.
syscall V1 GFX QueryFontEngine [345] {
    version: OUT c<uint32_t>;
    capabilities: OUT c<uint32_t>;
};

syscall V1 GFX BeginFontUpload [346] {
    totalBytes: IN c<uint32_t>;
    upload: OUT c<uint32_t>;
};

// Ordered, nonempty chunks of at most 4096 bytes; offset must equal bytes sent.
syscall V1 GFX AppendFontUpload [347] {
    upload: IN c<uint32_t>;
    offset: IN c<uint32_t>;
    data: IN buffer<c<uint8_t>>;
};

// Success consumes the upload and returns a new-generation immutable face
// handle. Invalid normalized data consumes/releases the upload and reports
// undos::font::Error. Incomplete uploads remain open and can be continued.
// Structural registration does not assert instruction execution support.
syscall V1 GFX CommitFontUpload [348] {
    upload: IN c<uint32_t>;
    face: OUT c<uint32_t>;
    validationError: OUT c<uint32_t>;
};

syscall V1 GFX CloseFontResource [349] {
    handle: IN c<uint32_t>;
};
