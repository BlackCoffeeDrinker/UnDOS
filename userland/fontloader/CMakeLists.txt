add_library(fontdecode STATIC decoder.cpp)
target_link_libraries(fontdecode PUBLIC libkcpp KernelUserlandShared)
target_include_directories(fontdecode PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(fontdecode PRIVATE -ffreestanding -nostdlib -fno-exceptions -fno-rtti -fno-pic -fno-pie -m32)

add_executable(fontloader main.cpp)
target_include_directories(fontloader PRIVATE ${CMAKE_SOURCE_DIR}/userland)
target_link_libraries(fontloader PRIVATE fontdecode undoscrt undos_font_resource)
set_target_properties(fontdecode fontloader PROPERTIES POSITION_INDEPENDENT_CODE OFF)
target_compile_options(fontloader PRIVATE -ffreestanding -nostdlib -fno-exceptions -fno-rtti -fno-pic -fno-pie -m32)
target_link_options(fontloader PRIVATE -nostdlib -no-pie -Wl,-T,${CMAKE_SOURCE_DIR}/userland/apps/term/linker.ld)
