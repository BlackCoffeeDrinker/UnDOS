// -----------------------------------------------------

using HANDLER = c<uint32_t>;
using COLORREF = c<uint32_t>;
// COLORREF is a palette16 index (0..15), shared with USER class backgrounds.
// Packed RGB targets receive the corresponding canonical VGA RGB color.
using POINT = struct<Point>;
using RECT  = struct<Rect>;


// -----------------------------------------------------
//  Flag Types
// -----------------------------------------------------

    Primary;
    Memory;
    Offscreen;
    NoAccel;
};

flags PenStyle : uint32 {
    Solid;
    Dash;
    Dot;
    DashDot;
    DashDotDot;
};

flags BrushPattern : uint32 {
    Solid;
    Hatch;
    Diagonal;
    Cross;
};

flags RegionCombineMode : uint32 {
    Union;
    Intersect;
    Subtract;
    Xor;
};

flags Device3DFlags : uint32 {
    Hardware;
    Software;
    VertexAccel;
    TextureAccel;
};


// -----------------------------------------------------
//  Struct Types

struct Point {
    x: c<int32_t>;
    y: c<int32_t>;
};

struct Rect {
    left:   c<int32_t>;
    top:    c<int32_t>;
    right:  c<int32_t>;
    bottom: c<int32_t>;
};

struct DeviceCaps {
    width:      c<uint32_t>;
    height:     c<uint32_t>;
    colorDepth: c<uint32_t>;
    accelFlags: c<uint32_t>;
};

/** Primary-display mode. modeId is an opaque driver mode ID, not the
 * enumeration ordinal. Refresh rate is in Hz; colorDepth is bits per pixel. */
struct DisplayMode {
    modeId:     c<uint32_t>;
    width:      c<uint32_t>;
    height:     c<uint32_t>;
    colorDepth: c<uint32_t>;
    refreshHz:  c<uint32_t>;
};

struct DisplayState {
    /** Zero before activation or after an output-enable failure. */
    active:     c<uint32_t>;
    /** Changes when a new mode is programmed, including enable failure. */
    generation: c<uint32_t>;
    /** All zero before the first mode is programmed. */
    mode:       struct<DisplayMode>;
};

// -----------------------------------------------------
//  Device Context & Object Lifecycle
// -----------------------------------------------------

// Zero/Primary selects the active screen; never implicitly programs a mode.
// Returns UNSUPPORTED until SetDisplayMode activates the display.
// Memory/Offscreen creates a DC with no target
// until SelectObject selects a bitmap. Primary cannot combine with these flags.
// NoAccel bypasses packed acceleration, never the required planar adapter path.
syscall V1 GFX CreateDC [300] {
    driverName: IN string;
    deviceName: IN string;
    initFlags:  IN flags<DCInitFlags>;
    handle:     OUT c<uint32_t>;
};

syscall V1 GFX DeleteDC [301] {
    dc: IN HANDLER;
};

// Only bitmap selection into Memory/Offscreen DCs is supported.
syscall V1 GFX SelectObject [302] {
    dc:     IN HANDLER;
    object: IN HANDLER;
};

syscall V1 GFX CreatePen [303] {
    style: IN flags<PenStyle>;
    width: IN c<uint32_t>;
    color: IN COLORREF;
};

syscall V1 GFX CreateBrush [304] {
    pattern: IN flags<BrushPattern>;
    color:   IN COLORREF;
};

// Retired font/text syscall tombstone. Do not reuse ID 305.

// Top-down packed bitmap, zero initialized; maximum backing allocation 16 MiB.
// INDEX8 uses palette16 indices; RGB24/ARGB32 stores native BGR/BGRA bytes.
syscall V1 GFX CreateBitmap [306] {
    width:   IN c<uint32_t>;
    height:  IN c<uint32_t>;
    format:  IN enum<PixelFormat>;
    pixels:  IN buffer<c<uint8_t>>;
    handle:  OUT c<uint32_t>;
};

syscall V1 GFX DeleteObject [307] {
    object: IN HANDLER;
};

    srcDC:     IN HANDLER;

// Ordered, nonempty chunks of at most 4096 bytes; offset must equal bytes sent.
syscall V1 GFX AppendFontUpload [347] {
    upload: IN c<uint32_t>;
    offset: IN c<uint32_t>;
    data: IN buffer<c<uint8_t>>;
};

// Success consumes the upload and returns a new-generation immutable face
// handle. Invalid normalized data consumes/releases the upload and reports
// undos::font::Error. Incomplete uploads remain open and can be continued.
// Structural registration does not assert instruction execution support.
syscall V1 GFX CommitFontUpload [348] {
    upload: IN c<uint32_t>;
    face: OUT c<uint32_t>;
    validationError: OUT c<uint32_t>;
};

syscall V1 GFX CloseFontResource [349] {
    handle: IN c<uint32_t>;
};

/** Enumerates monitor-approved primary-display modes without activating output.
 * Ordinals start at zero. OKAY with available=0 and a zero mode marks the end.
 * UNSUPPORTED means no usable primary display. */
syscall V1 GFX EnumDisplayModes [350] {
    index:     IN c<uint32_t>;
    mode:      OUT struct<DisplayMode>;
    available: OUT c<uint32_t>;
};

/** Queries the primary display, including its inactive pre-graphics state.
 * Returns UNSUPPORTED when no primary display exists. */
syscall V1 GFX GetDisplayMode [351] {
    state: OUT struct<DisplayState>;
};

/** Activates or changes the primary display. Any process may call this.
 * modeId comes from EnumDisplayModes, not its enumeration ordinal.
 * Existing screen DC handles remain valid; pixels are lost and all windows
 * repaint. DisplayChange is coalesced per window and survives a full queue.
 * Invalid/monitor-rejected IDs fail without changing the display. Allocation
 * failures preserve the previous mode. Hardware enable failure reports UNKNOWN;
 * GetDisplayMode then reports the programmed mode as inactive.
 * This call does not persist a preference. */
syscall V1 GFX SetDisplayMode [352] {
    modeId: IN c<uint32_t>;
};
