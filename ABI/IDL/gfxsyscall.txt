// -----------------------------------------------------

using HANDLER = c<uint32_t>;

// COLORREF is a palette16 index (0..15), shared with USER class backgrounds.
// Packed RGB targets receive the corresponding canonical VGA RGB color.
using POINT = struct<Point>;
using RECT  = struct<Rect>;

    x: c<int32_t>;
    y: c<int32_t>;
};

    left:   c<int32_t>;

    width:      c<uint32_t>;
    height:     c<uint32_t>;

/** One logical source mode. IDs are valid only for the reported generation. */
struct VideoPresentSourceMode {
    sourceId: c<uint32_t>;
    modeId: c<uint32_t>;
    width: c<uint32_t>;
    height: c<uint32_t>;
    colorDepth: c<uint32_t>;
    pixelFormat: c<uint32_t>;
};

/** One discovered physical output. Adaptive sync is descriptive only. */
struct VideoPresentTarget {
    targetId: c<uint32_t>;
    connected: c<uint32_t>;
    modeCount: c<uint32_t>;
    maxWidth: c<uint32_t>;
    maxHeight: c<uint32_t>;
    maxRefreshHz: c<uint32_t>;
    adaptiveSyncSupported: c<uint32_t>;
    adaptiveSyncMinMilliHz: c<uint32_t>;
    adaptiveSyncMaxMilliHz: c<uint32_t>;
};

/** A fixed-refresh target mode. */
struct VideoPresentTargetMode {
    targetId: c<uint32_t>;
    modeId: c<uint32_t>;
    width: c<uint32_t>;
    height: c<uint32_t>;
    colorDepth: c<uint32_t>;
    pixelFormat: c<uint32_t>;
    refreshHz: c<uint32_t>;
};

/** One source-to-target route. This phase accepts one untransformed path. */
struct VideoPresentPath {
    sourceId: c<uint32_t>;
    sourceModeId: c<uint32_t>;
    targetId: c<uint32_t>;
    targetModeId: c<uint32_t>;
    sourceX: c<int32_t>;
    sourceY: c<int32_t>;
    width: c<uint32_t>;
    height: c<uint32_t>;
    flags: c<uint32_t>;
};

/** Generation-tagged bounded topology summary. */
struct VideoPresentTopology {
    generation: c<uint32_t>;
    sourceCount: c<uint32_t>;
    sourceModeCount: c<uint32_t>;
    targetCount: c<uint32_t>;
    targetModeCount: c<uint32_t>;
    pathCount: c<uint32_t>;
    recordLimit: c<uint32_t>;
    degraded: c<uint32_t>;
};

enum VideoPresentOutcome {
    Success = 0,
    ProgrammingFailed = 1,
    EnableFailed = 2,
    StateUnknown = 3,
};

/** Result of a target programming attempt, including actual active state. */
struct VideoPresentTargetResult {
    targetId: c<uint32_t>;
    requestedModeId: c<uint32_t>;
    actualModeId: c<uint32_t>;
    /** 0 means disabled, 1 means active, and 2 means hardware state is unknown. */
    active: c<uint32_t>;
    outcome: enum<VideoPresentOutcome>;
    degraded: c<uint32_t>;
};

// -----------------------------------------------------
//  Device Context & Object Lifecycle
// -----------------------------------------------------

// Zero/Primary selects the active screen; never implicitly programs a mode.
// Returns UNSUPPORTED until a video-present path activates the desktop.
// Memory/Offscreen creates a DC with no target
// until SelectObject selects a bitmap. Primary cannot combine with these flags.
// NoAccel bypasses packed acceleration, never the required planar adapter path.
syscall V1 GFX CreateDC [300] {
    driverName: IN string;
    deviceName: IN string;
    initFlags:  IN flags<DCInitFlags>;
    handle:     OUT c<uint32_t>;
};

syscall V1 GFX DeleteDC [301] {
    dc: IN HANDLER;
};

// Only bitmap selection into Memory/Offscreen DCs is supported.
syscall V1 GFX SelectObject [302] {
    dc:     IN HANDLER;
    object: IN HANDLER;
};

syscall V1 GFX CreatePen [303] {
    style: IN flags<PenStyle>;
    width: IN c<uint32_t>;
    color: IN COLORREF;
};

syscall V1 GFX CreateBrush [304] {
    pattern: IN flags<BrushPattern>;
    color:   IN COLORREF;
};

// Retired font/text syscall tombstone. Do not reuse ID 305.

// Top-down packed bitmap, zero initialized; maximum backing allocation 16 MiB.
// INDEX8 uses palette16 indices; RGB24/ARGB32 stores native BGR/BGRA bytes.
syscall V1 GFX CreateBitmap [306] {
    width:   IN c<uint32_t>;
    height:  IN c<uint32_t>;
    format:  IN enum<PixelFormat>;
    pixels:  IN buffer<c<uint8_t>>;
    handle:  OUT c<uint32_t>;
};

syscall V1 GFX DeleteObject [307] {
    object: IN HANDLER;
};

// Ordered, nonempty chunks of at most 4096 bytes; offset must equal bytes sent.
syscall V1 GFX AppendFontUpload [347] {
    upload: IN c<uint32_t>;
    offset: IN c<uint32_t>;
    data: IN buffer<c<uint8_t>>;
};

// Success consumes the upload and returns a new-generation immutable face
// handle. Invalid normalized data consumes/releases the upload and reports
// undos::font::Error. Incomplete uploads remain open and can be continued.
// Structural registration does not assert instruction execution support.
syscall V1 GFX CommitFontUpload [348] {
    upload: IN c<uint32_t>;
    face: OUT c<uint32_t>;
    validationError: OUT c<uint32_t>;
};

syscall V1 GFX CloseFontResource [349] {
    handle: IN c<uint32_t>;
};

/** Reads the current topology summary. Every array class is limited to 64 records. */
syscall V1 GFX GetVideoPresentTopology [350] {
    topology: OUT struct<VideoPresentTopology>;
};

/** Queries or fills source modes pinned to one immutable generation.
 * queryOnly=1 requires zero capacity and returns required, written=0.
 * Fill requires capacity >= required; INVALID_SIZE reports required without partial writes.
 * Stale generation returns RETRY with required=written=0. */
syscall V1 GFX EnumVideoPresentSourceModes [351] {
    generation: IN c<uint32_t>;
    sourceId: IN c<uint32_t>;
    firstIndex: IN c<uint32_t>;
    queryOnly: IN c<uint32_t>;
    modes: OUT buffer<struct<VideoPresentSourceMode>>;
    required: OUT c<uint32_t>;
    written: OUT c<uint32_t>;
};

/** Queries or fills targets from one immutable generation.
 * queryOnly=1 requires zero capacity. Fill reports required on INVALID_SIZE without writes.
 * firstIndex equal to count succeeds with required=written=0. */
syscall V1 GFX EnumVideoPresentTargets [352] {
    generation: IN c<uint32_t>;
    firstIndex: IN c<uint32_t>;
    queryOnly: IN c<uint32_t>;
    targets: OUT buffer<struct<VideoPresentTarget>>;
    required: OUT c<uint32_t>;
    written: OUT c<uint32_t>;
};

/** Queries or fills fixed-refresh modes for a target from one immutable generation.
 * queryOnly must be 0 (fill) or 1 (zero-capacity count query).
 * Undersized fills return INVALID_SIZE and required with written=0, without partial writes. */
syscall V1 GFX EnumVideoPresentTargetModes [353] {
    generation: IN c<uint32_t>;
    targetId: IN c<uint32_t>;
    firstIndex: IN c<uint32_t>;
    queryOnly: IN c<uint32_t>;
    modes: OUT buffer<struct<VideoPresentTargetMode>>;
    required: OUT c<uint32_t>;
    written: OUT c<uint32_t>;
};

/** Queries or fills committed paths from one immutable generation.
 * queryOnly=1 requires zero capacity; queryOnly=0 fills exactly required records.
 * RETRY writes no records; INVALID_SIZE reports required without partial writes. */
syscall V1 GFX EnumVideoPresentPaths [354] {
    generation: IN c<uint32_t>;
    firstIndex: IN c<uint32_t>;
    queryOnly: IN c<uint32_t>;
    paths: OUT buffer<struct<VideoPresentPath>>;
    required: OUT c<uint32_t>;
    written: OUT c<uint32_t>;
};

/** Copies and validates a complete single-path topology for the supplied generation.
 * The returned process-local token owns that immutable candidate. */
syscall V1 GFX ValidateVideoPresentTopology [355] {
    generation: IN c<uint32_t>;
    paths: IN buffer<struct<VideoPresentPath>>;
    token: OUT c<uint32_t>;
};

/** Commits a validated topology; required is always 1 and written initially 0.
 * Stale RETRY takes precedence over capacity and consumes the token.
 * Current-token INVALID_SIZE or OUT_OF_MEMORY preserves reservations with no hardware change.
 * A hardware attempt returns actual state in the result rather than hiding failure. */
syscall V1 GFX CommitVideoPresentTopology [356] {
    token: IN c<uint32_t>;
    results: OUT buffer<struct<VideoPresentTargetResult>>;
    required: OUT c<uint32_t>;
    written: OUT c<uint32_t>;
};
