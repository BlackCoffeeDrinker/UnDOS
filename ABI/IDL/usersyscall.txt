
// region USER
//
// =====================================================
//  Window Manager (USER)
//  Syscalls 400–499
// =====================================================
//

enum WindowMessage {
    Null        = 0x0000,
    Create      = 0x0001,
    Destroy     = 0x0002,
    Move        = 0x0003,
    Size        = 0x0005,
    Activate    = 0x0006,
    SetFocus    = 0x0007,
    KillFocus   = 0x0008,
    Paint       = 0x000F,
    Close       = 0x0010,
    Quit        = 0x0012,
    /** Coalesced display change; wParam=bpp, lParam=height<<16 | width.
     * Query the video-present topology and paths for current state. */
    DisplayChange = 0x007E,
    /** Coalesced font generation change; wParam is the current generation. */
    FontChanged = 0x007D,
    NcHitTest   = 0x0084,
    NcMouseMove = 0x00A0,
    /** Single system-box click: wParam=SysMenu (3), lParam signed screen x/y.
     * Userland presents the system menu; system-box double-click sends Close.
     */
    NcLButtonDown = 0x00A1,
    NcLButtonUp   = 0x00A2,
    KeyDown     = 0x0100,
    KeyUp       = 0x0101,
    Char        = 0x0102,
    Command     = 0x0111,
    MouseMove   = 0x0200,
    LButtonDown = 0x0201,
    LButtonUp   = 0x0202,
    LButtonDblClk = 0x0203,
    RButtonDown = 0x0204,
    RButtonUp   = 0x0205,
    CaptureChanged = 0x0215,
    User        = 0x0400,
};

flags WindowStyle : uint32 {
    Visible;
    Caption;
    Border;
    Child;
};

enum ShowCommand {
    Hide = 0,
    Show = 1,
    Minimize = 2,
    Maximize = 3,
    Restore = 4,
};

enum HitTest {
    Transparent = -1,
    Nowhere = 0,
    Client = 1,
    Caption = 2,
    SysMenu = 3,
    MinButton = 8,
    MaxButton = 9,
    Left = 10,
    Right = 11,
    Top = 12,
    TopLeft = 13,
    TopRight = 14,
    Bottom = 15,
    BottomLeft = 16,
    BottomRight = 17,
    Border = 18,
    Close = 20,
};

enum WindowLongIndex {
    WndProc  = 0,
    UserData = 1,
    Style    = 2,
    /** Parent HWND, or zero for a root. Read-only. */
    Parent   = 3,
};

enum MenuItemFlags {
    String    = 0,
    Grayed    = 0x0001,
    Disabled  = 0x0002,
    Popup     = 0x0010,
    Separator = 0x0800,
};

struct Msg {
    hwnd:    c<uint32_t>;
    message: c<uint32_t>;
    wParam:  c<uint32_t>;
    lParam:  c<uint32_t>;
    time:    c<uint32_t>;
};

struct PaintInfo {
    /** Invalid area in client coordinates (half-open). */
    rcPaint: struct<Rect>;
    /** Non-zero when the background was erased by the kernel. */
    erased:  c<uint32_t>;
};

/** Registers a process-scoped window class; wndProc is opaque and called by userland. */
syscall V1 USER RegisterClass [400] {
    name:       IN string;
    wndProc:    IN c<uint32_t>;
    background: IN COLORREF;
};

/** Creates a window owned by the calling thread. Child requires a same-process
 * parent; x/y are relative to its client origin. Otherwise parent must be zero.
 * Coordinates and dimensions must fit the signed 16-bit screen range.
 */
syscall V1 USER CreateWindow [401] {
    className: IN string;
    title:     IN string;
    style:     IN flags<WindowStyle>;
    x:         IN c<int32_t>;
    y:         IN c<int32_t>;
    width:     IN c<int32_t>;
    height:    IN c<int32_t>;
    parent:    IN c<uint32_t>;
    userData:  IN c<uint32_t>;
    hwnd:      OUT c<uint32_t>;
};

/** Immediately destroys the window and descendants and removes their queued
 * messages. Userland dispatches any Destroy callback before this syscall;
 * no message may subsequently target an invalidated handle.
 */
syscall V1 USER DestroyWindow [402] {
    hwnd: IN c<uint32_t>;
};

/** Hide preserves state, Show preserves state, Restore restores saved geometry.
 * Minimize creates an icon plus label for children in the parent client
 * (roots retain a desktop caption). Click selects a child icon, double-click
 * or Enter restores it; Ctrl+Tab/F6 cycles including minimized child icons.
 * Maximize fills the parent client or desktop. Parent hide/minimize suppresses
 * descendant visibility.
 */
syscall V1 USER ShowWindow [403] {
    hwnd:    IN c<uint32_t>;
    command: IN c<uint32_t>;
};

syscall V1 USER MoveWindow [404] {
    hwnd:   IN c<uint32_t>;
    x:      IN c<int32_t>;
    y:      IN c<int32_t>;
    width:  IN c<int32_t>;
    height: IN c<int32_t>;
};

/** Window rectangle in screen coordinates. */
syscall V1 USER GetWindowRect [405] {
    hwnd: IN c<uint32_t>;
    rect: OUT RECT;
};

/** Adds rect (client coordinates) to the invalid area; whole != 0 invalidates the client. */
syscall V1 USER InvalidateRect [406] {
    hwnd:  IN c<uint32_t>;
    rect:  IN RECT;
    whole: IN c<uint32_t>;
};

/** Returns a DC clipped to the visible invalid client, origin at the client.
 * Only one paint DC per window; EndPaint must match its hwnd/DC pair.
 */
syscall V1 USER BeginPaint [407] {
    hwnd:  IN c<uint32_t>;
    dc:    OUT c<uint32_t>;
    paint: OUT struct<PaintInfo>;
};

syscall V1 USER EndPaint [408] {
    hwnd: IN c<uint32_t>;
    dc:   IN c<uint32_t>;
};

/** Blocks until a message arrives; msg.message == Quit ends a message loop. */
syscall V1 USER GetMessage [409] {
    msg: OUT struct<Msg>;
};

/** Non-blocking; returns TIMEOUT when the queue is empty. remove == 0 leaves it queued. */
syscall V1 USER PeekMessage [410] {
    msg:    OUT struct<Msg>;
    remove: IN c<uint32_t>;
};

/** Queues a message; returns OUT_OF_MEMORY when the target ring is full. */
syscall V1 USER PostMessage [411] {
    hwnd:    IN c<uint32_t>;
    message: IN c<uint32_t>;
    wParam:  IN c<uint32_t>;
    lParam:  IN c<uint32_t>;
};

/** One of WindowLongIndex. */
syscall V1 USER GetWindowLongPtr [412] {
    hwnd:  IN c<uint32_t>;
    index: IN c<uint32_t>;
    value: OUT c<uint32_t>;
};

syscall V1 USER SetWindowLongPtr [413] {
    hwnd:     IN c<uint32_t>;
    index:    IN c<uint32_t>;
    value:    IN c<uint32_t>;
    previous: OUT c<uint32_t>;
};

syscall V1 USER SetCapture [414] {
    hwnd: IN c<uint32_t>;
};

syscall V1 USER ReleaseCapture [415] {
    unused: IN c<uint32_t>;
};

/** Client bounds {0,0,width,height}; empty while minimized. */
syscall V1 USER GetClientRect [416] {
    hwnd: IN c<uint32_t>;
    rect: OUT RECT;
};

/** Focus a visible same-process window and activate its root. A minimized
 * child icon may receive focus for Enter/Alt+Space; minimized roots cannot.
 * hwnd == 0 clears this process's focus without changing activation.
 * Key lParam: bits 0..7 scan code, 8..10 key_flags, 16..24 modifier_flags.
 * Char lParam carries the same metadata. Mouse wParam low bits: left=1,right=2;
 * modifier_flags occupy bits 16..24. Mouse lParam packs signed client x/y.
 */
syscall V1 USER SetFocus [417] {
    hwnd: IN c<uint32_t>;
};

/** Set a same-process window title (at most 63 bytes plus terminator). */
syscall V1 USER SetWindowText [418] {
    hwnd: IN c<uint32_t>;
    text: IN string;
};

/** Creates a menu handle owned by the calling process. popup must be zero for
 * a menu bar or one for a popup menu.
 */
syscall V1 USER CreateMenu [419] {
    popup: IN c<uint32_t>;
    menu: OUT c<uint32_t>;
};

/** Destroys a process-owned menu and its submenu tree, and detaches it from
 * windows. A submenu cannot be destroyed separately while it belongs to a
 * parent menu.
 */
syscall V1 USER DestroyMenu [420] {
    menu: IN c<uint32_t>;
};

/** Appends a command, submenu, or separator to a menu owned by the caller.
 * Popup requires item to be an unparented popup menu owned by the caller.
 * Separator ignores item and text. Other items require nonempty text, limited
 * to 63 bytes. Supported flags are Grayed, Disabled, Popup, and Separator;
 * Grayed and Disabled cannot be combined with Separator.
 */
syscall V1 USER AppendMenu [421] {
    menu: IN c<uint32_t>;
    flags: IN enum<MenuItemFlags>;
    item: IN c<uint32_t>;
    text: IN string;
};

/** Attaches a menu bar owned by the caller to one of its windows. menu zero
 * detaches the current menu. Popup menus cannot be attached as menu bars.
 */
syscall V1 USER SetMenu [422] {
    hwnd: IN c<uint32_t>;
    menu: IN c<uint32_t>;
};

/** Retrieves the menu-bar handle attached to a window, or zero when none is
 * attached.
 */
syscall V1 USER GetMenu [423] {
    hwnd: IN c<uint32_t>;
    menu: OUT c<uint32_t>;
};

// endregion
