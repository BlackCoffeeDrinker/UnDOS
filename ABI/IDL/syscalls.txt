// The GFX (GDI-style) syscall group (300-399) lives in its own IDL file and
// is pulled in here so both groups end up in a single generated dispatch
// table (see kernel/src/gfx/gfx_management.cpp and kernel/src/syscalls/GFX.cpp).
include "gfxsyscall.txt";
include "usersyscall.txt";
include "abisyscall.txt";
include "netsyscall.txt";

/** Opaque 32-bit process-local object handle, not a pointer or a stream index.
 * Zero is a valid handle. Passing a handle to another process requires an
 * explicit inheritance or duplication operation; its numeric value may change.
 * Syscall status determines whether an output handle is valid. */
using HANDLE = c<uint32_t>;

enum SyscallReturnCode {
    OKAY = 0,
    UNKNOWN = 0xFFFFFFFF,
    INVALID_SIZE = 0xFFFFFFFE,
    OUT_OF_MEMORY = 0xFFFFFFFD,
    INVALID_ADDRESS = 0xFFFFFFFC,
    INVALID_ARGUMENT = 0xFFFFFFFB,
    UNSUPPORTED = 0xFFFFFFFA,
    TIMEOUT = 0xFFFFFFF9,

};

/** Flags controlling how a new process is created. */
enum ProcessFlags {
    /** No special behavior. */
    None        = 0,
    /** The child process is detached from its parent (no notification on exit). */
    Detached    = 1,
    /** The child process inherits the parent's environment variables. */
    InheritEnv  = 2,
};

syscall V1 DEBUG Print [100] {
    message: IN string;
};

/**
 * Creates a new process from the executable at `path`. Waits only for loader
 * mapping/dependency/relocation completion, not constructors or child exit.
 * There is no timeout: the bounded loader must signal readiness or exit on
 * failure, so failure never leaves a running child without an owned handle.
 * On failure no process handle is returned.
 * Successful handles may be closed immediately without terminating the child.
 */
syscall V1 PROC CreateProcess [200] {
    /** Nonempty executable path, at most 255 bytes; becomes argv[0] verbatim. */
    path: IN string;
    /** At most 1023 bytes, split on ASCII whitespace outside double quotes.
     * Quotes are removed, adjacent fragments concatenate, "" is an empty arg.
     * Backslashes are literal; unmatched quotes/NULs reject. Max argc is 32. */
    argument: IN string;
    /** Only ProcessFlags_None (zero) is supported; all other flags reject. */
    flags: IN c<uint32_t>;
    /** Handle to the new process (waitable with WaitProcess). */
    handle: OUT HANDLE;
};

/** Creates a process with explicit standard streams. Same path, argument,
 * flags, readiness and handle ownership contract as CreateProcessV1.
 * Only these streams are inherited; unrelated handles are not inherited.
 * Handles are caller-local open I/O objects, stdin requires read access,
 * stdout/stderr require write access. Child handles are independent copies. */
syscall V2 PROC CreateProcess [208] {
    path: IN string;
    argument: IN string;
    flags: IN c<uint32_t>;
    stdinHandle: IN HANDLE;
    stdoutHandle: IN HANDLE;
    stderrHandle: IN HANDLE;
    handle: OUT HANDLE;
};

/** Retrieves information about a running or exited process. */
syscall V1 PROC GetInfo [201] {
    /** Process ID to query. */
    pid: IN c<uint32_t>;
    /** Command line the process was started with (query-length-then-fill). */
    commandLine: OUT string;
    /** Exit code of the process, if it has exited. */
    exitCode: OUT c<uint32_t>;
};

/** Terminates the current process with the given return code. */
syscall V1 PROC Exit [202] {
    /** Return code to report to the parent process. */
    exitCode: IN c<uint32_t>;
};

/** Timeout value meaning "wait forever". */
enum WaitTimeout {
    Infinite = 0xFFFFFFFF,
};

/** Blocks until a waitable handle (event or process) is signalled. Returns TIMEOUT on expiry. */
syscall V1 PROC WaitForObject [203] {
    /** Handle to wait on. */
    handle: IN HANDLE;
    /** Timeout in milliseconds; 0 polls, WaitTimeout_Infinite never expires. */
    timeoutMs: IN c<uint32_t>;
};

/** Blocks until a process exits and returns its exit code. Returns TIMEOUT on expiry. */
syscall V1 PROC WaitProcess [204] {
    /** Process handle from CreateProcess. */
    handle: IN HANDLE;
    /** Timeout in milliseconds; 0 polls, WaitTimeout_Infinite never expires. */
    timeoutMs: IN c<uint32_t>;
    /** Exit code of the process. */
    exitCode: OUT c<uint32_t>;
};

/** Memory protection bit flags, mirrors kernel::vmm::ProtectFlags. */
enum MemoryProtectFlags {
    /** No access. */
    None       = 0,
    /** Page is readable. */
    Read       = 1,
    /** Page is writable. */
    Write      = 2,
    /** Page is executable. */
    Execute    = 4,
};

/** Allocates zeroed pages, rounding size up to 4096. Read is required; unknown
 * flags and no-access/write-only mappings reject. i386 does not enforce NX. */
syscall V1 MEM AllocatePages [210] {
    /** Number of bytes to allocate (rounded up to a page boundary). */
    size: IN c<uint32_t>;
    /** Protection to apply to the newly allocated pages. */
    flags: IN enum<MemoryProtectFlags>;
    /** Base address of the allocated region. */
    address: OUT c<uint32_t>;
};

/** Frees a whole eager anonymous region; base must be page aligned and size,
 * rounded up to 4096, must equal the region size. No partial frees or aliases. */
syscall V1 MEM FreePages [211] {
    /** Base address of the region to free. */
    address: IN c<uint32_t>;
    /** Size in bytes of the region to free. */
    size: IN c<uint32_t>;
};

/** Reserved: currently rejects with INVALID_ARGUMENT (no partial VAD updates). */
syscall V1 MEM Protect [212] {
    /** Base address of the region to reprotect. */
    address: IN c<uint32_t>;
    /** Size in bytes of the region to reprotect. */
    size: IN c<uint32_t>;
    /** New protection to apply. */
    flags: IN enum<MemoryProtectFlags>;
};

/** Maps zeroed anonymous user pages at a page-aligned address selected by the
 * caller. Read is required; unknown flags and no-access/write-only mappings
 * reject. i386 does not enforce NX. */
syscall V1 MEM MapFixedPages [213] {
    address: IN c<uint32_t>;
    size: IN c<uint32_t>;
    flags: IN enum<MemoryProtectFlags>;
};

// Large, variable-length buffers (file I/O, draw command lists, IPC
// messages, ...) use buffer<T>: the pointer+length is validated once and
// then mapped directly, avoiding a copy and the RAM pressure it would cause.

flags FileOpenMode : uint32_t {
    Read;
    Write;
    Append;
    Create;
    Truncate;
    Exclusive;
};

syscall V1 IO Open [220] {
    /** Path to the object to open. */
    path: IN string;
    /** Open mode (read, write, append). */
    mode: IN flags<FileOpenMode>;
    /** Handle to receive the opened object. */
    handle: OUT HANDLE;
};

syscall V1 IO Close [221] {
    /** Open handle to close. */
    handle: IN HANDLE;
};

/** Reads data from an open object handle into `data`. */
syscall V1 IO Read [222] {
    /** Open handle to read from. */
    handle: IN HANDLE;
    /** Buffer that receives the data read. */
    data: OUT buffer<c<uint8_t>>;
};

/** Writes data from `data` to an open object handle. */
syscall V1 IO Write [223] {
    /** Open handle to write to. */
    handle: IN HANDLE;
    /** Buffer containing the data to write. */
    data: IN buffer<c<uint8_t>>;
};

/** Writes bytes and reports short writes. V1 remains available unchanged. */
syscall V2 IO Write [226] {
    /** Open handle with write access. */
    handle: IN HANDLE;
    /** Bytes to write. The dispatcher rejects lengths above
     * UNDOS_MAX_MAPPED_BUFFER_SIZE (64 MiB) with INVALID_SIZE; the target may
     * accept fewer bytes. Use null with length zero for an empty write. */
    data: IN buffer<c<uint8_t>>;
    /** Actual bytes accepted, at most dataLength. Zero on handler failure;
     * undefined if dispatcher validation rejects the call before the handler. */
    transferred: OUT c<uint32_t>;
};

/** File or directory information. Attributes use the FAT attribute bits:
 * read-only=1, hidden=2, system=4, directory=16, archive=32.
 * Timestamps and long filenames are not supplied by this interface. */
struct FileInformation {
    size: c<uint64_t>;
    attributes: c<uint32_t>;
};

/** Enumerates one directory entry by zero-based index, excluding . and ...
 * Requires a readable directory handle. found=0 means end of directory.
 * name uses query-length-then-fill, including its terminating NUL.
 * Indices are not snapshots: restart enumeration after a mutation.
 * On insufficient capacity returns INVALID_SIZE without truncation. */
syscall V1 IO ReadDirectory [227] {
    handle: IN HANDLE;
    index: IN c<uint32_t>;
    name: OUT string;
    information: OUT struct<FileInformation>;
    found: OUT c<uint32_t>;
};

/** Returns information for an open VFS file/directory handle with read access. */
syscall V1 IO GetFileInformation [229] {
    handle: IN HANDLE;
    information: OUT struct<FileInformation>;
};

/** Copies one file to a nonexisting destination on the same mount.
 * Both paths must be absolute, at most 255 bytes, with FAT short names.
 * Open source entries and open directories block the operation.
 * Directory copies and cross-volume copies are unsupported.
 * Device failures can leave partial changes; no crash consistency is promised. */
syscall V1 IO CopyFile [230] {
    source: IN string;
    destination: IN string;
};

/** Moves a file, or renames a directory within the same parent.
 * Both paths must be absolute, at most 255 bytes, with FAT short names.
 * The destination must not exist and must be on the same mount.
 * Open source entries and open directories block the operation.
 * Cross-parent directory moves and cross-volume moves are unsupported.
 * Device failures can leave partial changes; no crash consistency is promised. */
syscall V1 IO MoveFile [231] {
    source: IN string;
    destination: IN string;
};

/** Deletes a file or empty directory at an absolute path of at most 255 bytes.
 * Requires FAT short names. Roots, . and .. are rejected.
 * Open source entries, open directories and read-only entries block deletion.
 * Recursive deletion is unsupported. Device failures can leave partial changes;
 * no crash consistency is promised. */
syscall V1 IO DeleteFile [232] {
    path: IN string;
};

/** Creates a directory at a nonexisting absolute path of at most 255 bytes.
 * Requires FAT short names and an existing parent directory.
 * Open directory handles on the volume block creation.
 * Device failures can leave partial changes; no crash consistency is promised. */
syscall V1 IO CreateDirectory [233] {
    path: IN string;
};

/** Device control codes understood by IO Control. */
enum IoControlCode {
    /** Display: output is struct<IoctlDisplayMode>. */
    DisplayQueryMode = 0x0100,
    /** Pointer: input is struct<IoctlMouseConfig>. */
    MouseConfigure   = 0x0200,
    /** Gamepad: output is struct<IoctlGamepadState>. */
    GamepadReadState = 0x0300,
    /** Gamepad: output is struct<IoctlGamepadCaps>. */
    GamepadQueryCaps = 0x0301,
};

struct IoctlDisplayMode {
    width: c<uint32_t>;
    height: c<uint32_t>;
    bpp: c<uint32_t>;
    pitch: c<uint32_t>;
};

struct IoctlMouseConfig {
    /** Non-zero enables data reporting, zero disables it. */
    enable: c<uint32_t>;
    /** Sample rate in reports per second (10-200), 0 keeps the current rate. */
    sampleRate: c<uint32_t>;
};

/**
 * Latest axis/button snapshot of one gamepad or analog joystick. Axes are
 * normalized to -32767..32767 and centred on 0, independent of the device's
 * native resolution, and are named after their HID Generic Desktop usages.
 * The axes are spelled out one field at a time rather than as an array
 * because the IDL only supports array fields inside syscall bodies.
 */
struct IoctlGamepadState {
    /** Generic Desktop X. */
    leftX: c<int16_t>;
    /** Generic Desktop Y. */
    leftY: c<int16_t>;
    /** Generic Desktop Rx. */
    rightX: c<int16_t>;
    /** Generic Desktop Ry. */
    rightY: c<int16_t>;
    /** Generic Desktop Z. */
    leftTrigger: c<int16_t>;
    /** Generic Desktop Rz. */
    rightTrigger: c<int16_t>;
    /** Bit n is set while button n is down. */
    buttons: c<uint32_t>;
    /** D-pad / POV hat, each -1, 0 or +1. */
    hatX: c<int8_t>;
    hatY: c<int8_t>;
    /** Increments on every sample, so a reader can tell it skipped one. */
    sequence: c<uint32_t>;
};

/** Static capabilities of one gamepad or analog joystick. */
struct IoctlGamepadCaps {
    /** Number of axes the device actually reports. */
    axisCount: c<uint32_t>;
    /** Number of buttons the device actually reports. */
    buttonCount: c<uint32_t>;
    /** Non-zero when the device has a d-pad / POV hat. */
    hasHat: c<uint32_t>;
};

/** Sends a control code to the device behind an open handle. */
syscall V1 IO Control [224] {
    /** Open handle to the device. */
    handle: IN HANDLE;
    /** One of IoControlCode. */
    code: IN c<uint32_t>;
    /** Code-specific input payload. */
    input: IN buffer<c<uint8_t>>;
    /** Code-specific output payload; receives the number of bytes written. */
    output: OUT buffer<c<uint8_t>>;
};

/** Identifies one of the calling process's standard streams. */
enum StdHandleSelector {
    Stdin = 0,
    Stdout = 1,
    Stderr = 2,

/** Returns the opaque handle assigned to a standard stream, or
 * INVALID_ARGUMENT if that stream is not available to the calling process.

syscall V1 IO GetStdHandle [225] {
    /** One of StdHandleSelector. */
    selector: IN enum<StdHandleSelector>;
    /** Process-local opaque handle for the selected stream. */
    handle: OUT HANDLE;
};
